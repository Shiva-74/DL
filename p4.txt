
import torch
import torch.nn as nn
import torchvision
from torch.utils.data import DataLoader
from torchvision import transforms

class CNN(nn.Module):
    def __init__(self, n):
        super().__init__()
        self.conv = nn.Sequential(
            nn.Conv2d(3, 32, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
            nn.Conv2d(32, 64, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
            nn.Conv2d(64, 128, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2)
        )
        self.fc = nn.Sequential(nn.Flatten(), nn.Linear(2048, 256),
                                 nn.ReLU(), nn.Linear(256, n))

    def forward(self, x):
        return self.fc(self.conv(x))

device = "cuda" if torch.cuda.is_available() else "cpu"
tf = transforms.ToTensor()

for ds, n in [(torchvision.datasets.CIFAR10, 10),
              (torchvision.datasets.CIFAR100, 100)]:

    train = DataLoader(ds("./data", train=True, download=True, transform=tf), batch_size=128, shuffle=True)
    test = DataLoader(ds("./data", train=False, download=True, transform=tf), batch_size=128)

    model = CNN(n).to(device)
    opt = torch.optim.Adam(model.parameters(), lr=0.001)
    loss_fn = nn.CrossEntropyLoss()

    print("\nTraining:", ds.__name__)

    for e in range(10):
        model.train()
        for x, y in train:
            x, y = x.to(device), y.to(device)
            opt.zero_grad()
            loss = loss_fn(model(x), y)
            loss.backward()
            opt.step()

        model.eval()
        correct = total = 0
        with torch.no_grad():
            for x, y in test:
                x, y = x.to(device), y.to(device)
                correct += (model(x).argmax(1) == y).sum().item()
                total += y.size(0)

        print("Epoch:", e+1, "Accuracy:", round(correct/total*100, 2), "%")

print("Lab 4 complete")
